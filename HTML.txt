  <script>
    function updatePreview() {
      document.getElementById('pv-name').innerText = document.getElementById('fullName').value || 'Your Name';
      document.getElementById('pv-title').innerText = document.getElementById('jobTitle').value || 'Target Job Title';
      document.getElementById('pv-email').innerText = document.getElementById('email').value || 'email@domain.com';
      document.getElementById('pv-phone').innerText = document.getElementById('phone').value || '+63 900 000 0000';
      document.getElementById('pv-address').innerText = document.getElementById('address').value || 'City, Province';
      document.getElementById('pv-summary').innerText = document.getElementById('summary').value;
      document.getElementById('pv-company').innerText = document.getElementById('expCompany').value;
      document.getElementById('pv-role').innerText = document.getElementById('expRole').value;
      document.getElementById('pv-dates').innerText = document.getElementById('expDates').value;

      const bulletsText = document.getElementById('expBullets').value;
      const bulletsArray = bulletsText.split('\n').filter(b => b.trim() !== '');
      const bulletsList = document.getElementById('pv-bullets');
      bulletsList.innerHTML = '';
      bulletsArray.forEach(b => {
        const li = document.createElement('li');
        li.innerText = b;
        bulletsList.appendChild(li);
      });
    }

    function aiPolishSummary() {
      const summaryElem = document.getElementById('summary');
      summaryElem.value = "Accomplished Customer Service Professional with a strong track record of delivering exceptional client support, driving high CSAT ratings, and managing account resolutions in fast-paced environments. Proficient in CRM platforms and communication tools.";
      updatePreview();
      alert("Summary polished with professional phrasing!");
    }

    function triggerDownload() {
      const element = document.getElementById('resume-preview');
      const name = document.getElementById('fullName').value || 'Resume';
      const opt = {
        margin:       0.5,
        filename:     `${name.replace(/\s+/g, '_')}_Resume.pdf`,
        image:        { type: 'jpeg', quality: 0.98 },
        html2canvas:  { scale: 2 },
        jsPDF:        { unit: 'in', format: 'letter', orientation: 'portrait' }
      };
      html2pdf().set(opt).from(element).save();
    }

    // Initialize on page load
    window.onload = updatePreview;
  </script>
